def setup_logging(service_name: str) -> logging.Logger:
    formatter = jsonlogger.JsonFormatter(
        "%(levelname)s %(name)s %(message)s",
        rename_fields={
            "levelname": "level",
            "name": "service",
            "message": "event",
        },
    )
    handler = logging.StreamHandler()
    handler.setFormatter(formatter)
    root = logging.getLogger()
    root.handlers.clear()
    root.addHandler(handler)
    root.setLevel(logging.INFO)
    for name in ("uvicorn", "uvicorn.error", "uvicorn.access"):
        uv_logger = logging.getLogger(name)
        uv_logger.handlers.clear()
        uv_logger.propagate = False
        uv_logger.disabled = True
    return logging.getLogger(service_name)

Листинг В.1 - Тело главного цикла моделирования (loop.py)

def _run(self) -> None:
    client = self._find_client()
    while not self._stop_event.is_set():
        if self.state.ff_pending() > 0:
            self._drain_fast_forward(client)
            continue
        tick_started = time.perf_counter()
        self.state.advance_time(config.TICK_REAL_SEC)
        if self.state.running:
            now = self.state.virtual_time
            for sub in self.subsystems:
                sub.tick(now)
            if client is not None:
                client.flush()
        elapsed = time.perf_counter() - tick_started
        sleep_left = config.TICK_REAL_SEC - elapsed
        if sleep_left > 0:
            time.sleep(sleep_left)

Листинг В.2 - Генерация сенсорных показаний в штатном режиме

def _generate_sensor_readings(self, machine, product_code):
    profile = config.SENSOR_PROFILES.get(machine.machine_type, {})
    product_mods = config.SENSOR_MODIFIERS_BY_PRODUCT.get(product_code, {}) if product_code else {}
    result = {}
    for name, (mean, std, _unit) in profile.items():
        value = config.bounded_gauss(mean, std)
        product_mult = product_mods.get(name)
        if product_mult is not None:
            value *= product_mult
        anomaly_mult = machine.anomaly_modifier.get(name)
        if anomaly_mult is not None:
            value *= anomaly_mult
        result[name] = round(value, 4)
    return result

def _advance_phase(self, machine, load, now):
    duration_sec = config.FURNACE_PHASE_DURATIONS_SEC[load.phase]
    elapsed = (now - load.phase_started_at).total_seconds()
    if elapsed < duration_sec:
        return
    old_phase = load.phase
    next_phase = config.FURNACE_NEXT_PHASE[old_phase]
    load.phase = next_phase
    load.phase_started_at = now

Листинг Г.1 - Подготовка признаков (features.py)

def machine_frames(long_df, resample=None):
    resample = resample or config.RESAMPLE_RULE
    frames = {}
    if long_df.empty:
        return frames
    for mid, g in long_df.groupby("machine_id"):
        mtype = g["machine_type"].iloc[0]
        if mtype in config.SKIP_MACHINE_TYPES:
            continue
        if mtype == "furnace":
            raw = g.pivot_table(index="event_time", columns="sensor",
                                values="value", aggfunc="mean").sort_index()
            for phase in config.FURNACE_ML_PHASES:
                pw = filter_furnace_phase(raw, phase)
                if pw.empty:
                    continue
                pw = pw.resample(resample).mean().dropna(axis=1, how="all")
                pw = pw.ffill().dropna(how="all")
                if not pw.empty:
                    frames[(mid, phase)] = (mtype, pw)
        else:
            for pcode, pg in g.groupby("product_code", dropna=True):
                if pd.isna(pcode) or pcode is None:
                    continue
                wide = pg.pivot_table(index="event_time", columns="sensor",
                                      values="value", aggfunc="mean").sort_index()
                wide = wide.resample(resample).mean().ffill().dropna(how="all")
                if wide.empty:
                    continue
                frames[(mid, pcode)] = (mtype, wide)
    return frames

Листинг Г.2 - Обучение детектора и калибровка порога (detectors.py)

def fit(self, wide, machine_type=None):
    X = wide.dropna()
    self.columns = list(wide.columns)
    if len(X) < config.MIN_TRAIN_POINTS or X.shape[1] == 0:
        return False
    if machine_type:
        self.machine_type = machine_type
    self.ecod = ECOD(contamination=self.contamination)
    self.ecod.fit(X.values)
    self.iforest = IForest(contamination=self.contamination,
                           random_state=config.IFOREST_RANDOM_STATE)
    self.iforest.fit(X.values)
    if config.THRESHOLD_SIGMA > 0:
        s = self.ecod.decision_scores_
        self.ecod.threshold_ = s.mean() + config.THRESHOLD_SIGMA * s.std()
        s = self.iforest.decision_scores_
        self.iforest.threshold_ = s.mean() + config.THRESHOLD_SIGMA * s.std()
    self.train_mean = X.mean()
    self.train_std = X.std(ddof=0).replace(0, np.nan)
    _bin = pd.Timedelta(config.RESAMPLE_RULE).total_seconds()
    _n = max(1.0, _bin / config.SENSOR_INTERVAL_SEC)
    self.train_std_resampled = self.train_std / np.sqrt(_n)
    self.n_train = len(X)
    return True

Листинг Г.3 - Скоринг и комбинированное решающее правило (detectors.py)

def score(self, wide):
    if not self.fitted:
        return None
    X = wide.reindex(columns=self.columns).ffill().dropna()
    if X.empty:
        return None
    res = pd.DataFrame(index=X.index)
    res["label_ecod"]    = self.ecod.predict(X.values).astype(int)
    res["label_iforest"] = self.iforest.predict(X.values).astype(int)
    z = (X - self.train_mean) / self.train_std
    z = z.abs().replace([np.inf, -np.inf], np.nan).fillna(0.0)
    res["top_sensor"]   = z.idxmax(axis=1)
    res["top_sensor_z"] = z.max(axis=1).round(3)
    if config.ANOMALY_Z > 0:
        z_flag = res["top_sensor_z"] >= config.ANOMALY_Z
    else:
        z_flag = pd.Series(False, index=res.index)
    if config.ANOMALY_COMBINE == "both":
        det_flag = (res["label_ecod"] == 1) & (res["label_iforest"] == 1)
    else:
        det_flag = (res["label_ecod"] == 1) | (res["label_iforest"] == 1)
    res["is_anomaly"] = z_flag | det_flag
    return res

Листинг Г.4 - Правило персистентности (pipeline.py)

def _apply_persistence(self, key, scored):
    n = config.PERSIST_N
    if n <= 1:
        return scored
    counts = self._persist_count.get(key, {})
    flags = []
    for hot in scored["hot"].tolist():
        counts = {ch: counts.get(ch, 0) + 1 for ch in hot}
        flags.append(any(c >= n for c in counts.values()))
    self._persist_count[key] = counts
    scored["is_anomaly"] = flags
    return scored

Листинг Г.5 - Предиктивный контур: прогноз и проверка выхода за норму (pipeline.py)

def _prophet_check_sensor(self, mid, mtype, ctx, det, wide, sensor):
    ser = wide[sensor].dropna()
    if len(ser) < config.PROPHET_MIN_POINTS:
        return None
    fc = forecaster.forecast_series(ser, config.FORECAST_HORIZON_MIN)
    if fc is None:
        return None
    mean = float(det.train_mean[sensor])
    std = float(det.train_std[sensor])
    band_std = self._prophet_band_std(det, sensor, std)
    lower = mean - config.ANOMALY_Z * band_std
    upper = mean + config.ANOMALY_Z * band_std
    future = fc[fc["actual"].isna()]
    over = (future["yhat"] < lower) | (future["yhat"] > upper)
    n_breaches = int(over.sum())
    is_anom = n_breaches > 0
    now_v = fc.loc[fc["actual"].notna(), "ts"].max()
    lead_min = None
    if is_anom:
        first_ts = future.loc[over, "ts"].min()
        lead_min = round((first_ts - now_v).total_seconds() / 60.0, 1)
    return (mid, mtype, ctx, sensor, is_anom, n_breaches, int(len(future)),
            lead_min, round(lower, 4), round(upper, 4),
            round(float(future["yhat"].iloc[-1]), 4), now_v)

Листинг Г.6 - Постраничный запрос к Loki (loki_client.py)

def query_range(logql, start, end, limit=None, max_lines=None):
    limit = limit or config.LOKI_PAGE_LIMIT
    max_lines = max_lines or config.MAX_LOG_LINES
    url = f"{config.LOKI_URL}/loki/api/v1/query_range"
    collected = []
    cur_end = end
    while True:
        params = {"query": logql, "start": _ns(start), "end": _ns(cur_end),
                  "limit": limit, "direction": "backward"}
        r = requests.get(url, params=params, timeout=config.LOKI_TIMEOUT_SEC)
        r.raise_for_status()
        result = r.json().get("data", {}).get("result", [])
        page = []
        for stream in result:
            for ns, line in stream.get("values", []):
                page.append((int(ns), line))
        if not page:
            break
        page.sort(key=lambda x: x[0], reverse=True)
        collected.extend(page)
        if len(collected) >= max_lines:
            collected = collected[:max_lines]
            break
        oldest_ns = page[-1][0]
        new_end = datetime.fromtimestamp(oldest_ns / 1e9, tz=timezone.utc) - timedelta(microseconds=1)
        if len(page) < limit or new_end <= start:
            break
        cur_end = new_end
    records = []
    for _ns_ts, line in collected:
        try:
            records.append(json.loads(line))
        except (json.JSONDecodeError, TypeError):
            continue
    return records
